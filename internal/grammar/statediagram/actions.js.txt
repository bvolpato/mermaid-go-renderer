function processId() {
    const idx = yytext.indexOf('%%');
    if (idx === 0) {
        return false;
    }
    if (idx > 0) {
        const before = yytext.slice(0, idx);
        const after = yytext.slice(idx);

        if (after) {
            yy.lexer.unput(after);
        }
        yytext = before;
    }
    return true;
}
